function limpiarRutaImagen(ruta) { if (!ruta) return ''; if (ruta.startsWith('http://') || ruta.startsWith('https://')) return ruta; let limpia = ruta.replace(/^public\//, '').replace(/^\/+/, ''); return '/' + limpia; } async function cargarConfiguracionWeb() { const timestamp = Date.now(); const urls = [ `/contenido/configuracion.json?t=${timestamp}`, `https://raw.githubusercontent.com/${repoUsuario}/${repoNombre}/main/contenido/configuracion.json?t=${timestamp}` ]; for (let url of urls) { try { const res = await fetch(url); if (res.ok) { const texto = await res.text(); let config; try { config = JSON.parse(texto); } catch (err) { console.warn("Error interpretando JSON, intentando nuevamente..."); continue; } // Aplicar Logo if (config.brand && config.brand.logo) { const logoElem = document.getElementById('img-logo'); if (logoElem) logoElem.src = limpiarRutaImagen(config.brand.logo); } // Aplicar Banner y Textos if (config.hero) { if (config.hero.banner) { const bannerElem = document.getElementById('img-banner'); if (bannerElem) bannerElem.src = limpiarRutaImagen(config.hero.banner); } if (config.hero.titulo) { const titElem = document.getElementById('hero-titulo'); if (titElem) titElem.innerText = config.hero.titulo; } if (config.hero.subtitulo) { const subElem = document.getElementById('hero-subtitulo'); if (subElem) subElem.innerText = config.hero.subtitulo; } } // Aplicar Tipografías Dinámicas if (config.typography) { if (config.typography.font_headers) { document.documentElement.style.setProperty('--font-serif', `'${config.typography.font_headers}', serif`); } if (config.typography.font_body) { document.documentElement.style.setProperty('--font-sans', `'${config.typography.font_body}', sans-serif`); } } break; // Detener bucle tras carga exitosa } } catch (e) { console.warn("Fallo al obtener configuración desde:", url); } } }
